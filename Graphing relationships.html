Question 1

a. Import that data into R. Create a new data frame by pivoting the data into long format, creating two new variables, one for year, and the other for either Democratic or Republican vote proportion.

Long_data_frame <- State_Prez_vote |>
  pivot_longer(
    cols = matches("(obama|mccain|romney|clinton|trump|biden|harris)vote\\d+"),
    names_to = "candidate_year",
    values_to = "votes"
  ) |>
  separate(candidate_year, into = c("candidate", "year"), sep ="(?<=\\D)(?=\\d+)") |>
  mutate(
    party = case_match(candidate,
      c("obamavote", "clintonvote","bidenvote", "harrisvote")~ "Democratic",
      c("mccainvote","romneyvote","trumpvote") ~ "Republican"
    )
  )|>
  group_by(state,year) |>
  mutate(
    total_votes = sum(votes, na.rm =TRUE),
    vote_proportion = votes/total_votes
  )|>
  ungroup() |>
  select(state, year, party, candidate, vote_proportion) |>
  as_tibble()

a. Using the data from 1, generate a vote swing scatter plot with the 2024 Trump proportion of the vote on the Y axis and the 2020 Trump proportion of the vote on the X axis, labeling the points with the state names, and adding a 45 degree line to the plot.

trump_plot <- State_Prez_vote |>
  mutate(
    total_2020 = bidenvote20 + trumpvote20,
    total_2024 = trumpvote24 + Harrisvote24,
    trump_prop_2020 = trumpvote20 / total_2020,
    trump_prop_2024 = trumpvote24 / total_2024
  ) |>
  select(state, trump_prop_2020, trump_prop_2024) |>
  drop_na()

ggplot(trump_plot, aes(x = trump_prop_2020, y= trump_prop_2024)) +
  geom_point()+
  geom_text(aes(label = state), size = 2, vjust = 2) +
  geom_abline(slope = 1, intercept = 0, linetype = "dashed", color = "gray40") +
  labs(
    x = "Trump 2020 proportion of votes",
    y = "Trump 2024 proportion of votes",
    title = "Trump Proportion of votes 2020 vs 2024"
  )

The 45-degree dashed line represents a reference of change in Trump’s vote shares between 2020 and 2024. The points below the line show which states had less votes in 2024 and the points above the line show which states had lower votes in 2024.

trump_change <- State_Prez_vote |>
  mutate(
    total_2020 = trumpvote20 + bidenvote20,
    total_2024 = trumpvote24 + Harrisvote24,
    trump_prop_2020 = trumpvote20 / total_2020,
    trump_prop_2024 = trumpvote24 / total_2024,
    trump_change = trump_prop_2024 - trump_prop_2020
  ) |>
  select(state, trump_change)

us_map_data <- US_Map |>
  left_join(trump_change, by = c("NAME" = "state"))
ggplot(us_map_data) +
  geom_sf(aes(fill = trump_change), color = "black") +
  scale_fill_gradient2(
    low = "blue",  # Blue for decreases
    mid = "white",
    high = "red", # Red for increases
    midpoint = 0,
    name = "Change in\nTrump Vote Proportion"
  ) +
  labs(
    title = "Change in Trump Vote Share by State (2020 to 2024)"
  ) +
  theme_minimal()
